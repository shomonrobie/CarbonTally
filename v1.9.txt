CarbonTally Enterprise Migration: Principal Architect Review & Improvement
Code Review Report
Critical Issues Found:
Authentication Tables: Duplicate Supabase Auth functionality (sessions, tokens, MFA)

Subscription Design: Coupled to organizations table instead of proper billing architecture

RBAC: Permissions stored as JSON without proper normalization

Audit Log: Missing request tracing and API debugging fields

Validation Engine: Over-reliance on JSON for rule storage

Materialized Views: Trigger-based refresh causing performance issues

File Management: No centralized file tracking

Calculation History: No traceability for emissions calculations

Report Exports: Missing export tracking

Notifications: No preference system

Organization Settings: Mixed concerns in organizations table

Migration Strategy: Single monolithic migration

-- 9. Migrate existing organization_members to new schema
ALTER TABLE organization_members 
ADD COLUMN IF NOT EXISTS user_role_id UUID REFERENCES user_roles(id);

DO $$
DECLARE
    org_member RECORD;
    role_admin UUID;
    role_staff UUID;
    new_role_id UUID;
BEGIN
    SELECT id INTO role_admin FROM roles WHERE name = 'administrator';
    SELECT id INTO role_staff FROM roles WHERE name = 'analyst';
    
    FOR org_member IN SELECT * FROM organization_members LOOP
        -- Determine role mapping
        IF org_member.role = 'admin' THEN
            new_role_id := role_admin;
        ELSIF org_member.role = 'staff' THEN
            new_role_id := role_staff;
        ELSE
            new_role_id := (SELECT id FROM roles WHERE name = 'viewer');
        END IF;
        
        -- Create user role
        INSERT INTO user_roles (
            user_id,
            organization_id,
            role_id,
            granted_by,
            created_at
        ) VALUES (
            org_member.user_id,
            org_member.organization_id,
            new_role_id,
            org_member.user_id,
            NOW()
        );
    END LOOP;
END $$;

-- 10. Update existing staff_profiles
CREATE OR REPLACE FUNCTION migrate_staff_to_roles()
RETURNS VOID AS $$
DECLARE
    staff_record RECORD;
    role_id UUID;
BEGIN
    FOR staff_record IN SELECT * FROM staff_profiles LOOP
        -- Find matching role
        SELECT id INTO role_id FROM roles WHERE name = staff_record.role;
        
        -- Update staff profile with role_id
        UPDATE staff_profiles 
        SET role_id = role_id 
        WHERE id = staff_record.id;
    END LOOP;
END;
$$ LANGUAGE plpgsql;

SELECT migrate_staff_to_roles();
DROP FUNCTION migrate_staff_to_roles();

-- =====================================================
-- MIGRATION 030: MATERIALIZED VIEWS (Optimized)
-- =====================================================
-- Purpose: Performance optimization without triggers
-- Dependencies: None
-- Rollback: See 030_materialized_views_rollback.sql
-- =====================================================

BEGIN;

-- 1. Organization Emissions Summary
DROP MATERIALIZED VIEW IF EXISTS mv_organization_emissions_summary;
CREATE MATERIALIZED VIEW IF NOT EXISTS mv_organization_emissions_summary AS
SELECT 
    organization_id,
    DATE_TRUNC('month', start_date) as month,
    SUM(calculated_kg_co2e) as total_emissions_kg,
    SUM(calculated_kg_co2e) / 1000 as total_emissions_tonnes,
    COUNT(*) as transaction_count,
    COUNT(DISTINCT asset_id) as asset_count,
    MIN(start_date) as first_transaction,
    MAX(start_date) as last_transaction
FROM emissions_logs
GROUP BY organization_id, DATE_TRUNC('month', start_date)
WITH NO DATA;

-- 2. Organization Annual Summary
DROP MATERIALIZED VIEW IF EXISTS mv_organization_annual_summary;
CREATE MATERIALIZED VIEW IF NOT EXISTS mv_organization_annual_summary AS
SELECT 
    organization_id,
    EXTRACT(YEAR FROM start_date) as year,
    SUM(calculated_kg_co2e) as total_emissions_kg,
    SUM(calculated_kg_co2e) / 1000 as total_emissions_tonnes,
    COUNT(*) as transaction_count,
    AVG(calculated_kg_co2e) as avg_emission_per_transaction,
    MAX(calculated_kg_co2e) as max_single_emission
FROM emissions_logs
GROUP BY organization_id, EXTRACT(YEAR FROM start_date)
WITH NO DATA;

-- 3. Organization Daily Summary (for performance)
DROP MATERIALIZED VIEW IF EXISTS mv_organization_daily_summary;
CREATE MATERIALIZED VIEW IF NOT EXISTS mv_organization_daily_summary AS
SELECT 
    organization_id,
    start_date,
    SUM(calculated_kg_co2e) as daily_emissions_kg,
    COUNT(*) as daily_transactions
FROM emissions_logs
GROUP BY organization_id, start_date
WITH NO DATA;

    REFRESH MATERIALIZED VIEW CONCURRENTLY mv_organization_emissions_summary;
    REFRESH MATERIALIZED VIEW CONCURRENTLY mv_organization_annual_summary;
END;
