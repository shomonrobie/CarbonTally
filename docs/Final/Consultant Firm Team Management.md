Database Schema - Consultant Team (Phase 2)
sql
-- ============================================
-- CONSULTANT TEAM MANAGEMENT (PHASE 2)
-- ============================================

-- Consultant Firm
CREATE TABLE consultant_firms (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(255) NOT NULL,
    owner_id UUID NOT NULL REFERENCES users(id),
    -- Billing
    subscription_tier VARCHAR(50) DEFAULT 'consultant_pro',
    subscription_status VARCHAR(50) DEFAULT 'active',
    stripe_customer_id VARCHAR(255),
    stripe_subscription_id VARCHAR(255),
    -- Timestamps
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Consultant Firm Members (Team)
CREATE TABLE consultant_firm_members (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    firm_id UUID NOT NULL REFERENCES consultant_firms(id),
    user_id UUID NOT NULL REFERENCES users(id),
    role VARCHAR(50) NOT NULL, -- 'owner', 'consultant', 'analyst', 'intern'
    -- Permissions
    can_manage_clients BOOLEAN DEFAULT FALSE,
    can_upload_documents BOOLEAN DEFAULT TRUE,
    can_generate_reports BOOLEAN DEFAULT TRUE,
    can_manage_team BOOLEAN DEFAULT FALSE,
    -- Client access (NULL = all clients)
    client_access UUID[], -- Array of client IDs they can access
    -- Status
    is_active BOOLEAN DEFAULT TRUE,
    invited_by UUID REFERENCES users(id),
    invited_at TIMESTAMPTZ DEFAULT NOW(),
    joined_at TIMESTAMPTZ,
    -- Timestamps
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(firm_id, user_id)
);

-- Consultant Clients (Same as before)
CREATE TABLE consultant_clients (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    firm_id UUID NOT NULL REFERENCES consultant_firms(id),
    organization_id UUID NOT NULL REFERENCES organizations(id),
    -- Client details
    client_name VARCHAR(255) NOT NULL,
    client_industry VARCHAR(100),
    client_contact_email VARCHAR(255),
    -- Assigned consultant
    assigned_to UUID REFERENCES consultant_firm_members(id),
    -- Status
    status VARCHAR(20) DEFAULT 'active',
    -- Timestamps
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(firm_id, organization_id)
);
